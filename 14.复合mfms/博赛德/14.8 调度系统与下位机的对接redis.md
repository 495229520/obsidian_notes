# redis stream设计

> [!abstract] 小车私有redis层级表
> 通过小车唯一编码作为最上级key层级来进行区分，里面存放小车实时状态写入时间戳，调度系统下发指令信息stream，小车实时状态信息
> 只有robotCode：stream 用的是stream的结构，其他都是用的string的结构

```json
hyrms001                              // 复合机器人id,hyrms<3位id>
├── last_seen                         // 上一次写入实时状态的时间戳
├── state                             // 当前业务状态 "running" "pause" "cancel" "idle" "complete" "abort" "unknown"
├── task                              // 当前任务状态
│   └── id                            // 当前执行的任务 ID
│
├── waybill                           // 当前运单信息
│   ├── id                            // 当前执行的运单 ID
│   ├── type                          // 运单类型
│   ├── station                       // 运单目标站点唯一编码
│   ├── container_code                // 操作的货物 ID
│ 	└── sequence                      // 任务中第几个运单
│
├── command                           // 当前指令信息
│   ├── id                            // 当前成功执行的指令 ID
│   └── action                        // 当前成功执行的指令 action
│
└── agv                               // 小车实时状态
    ├── name                          // 小车别名
    ├── err_code                      // 当前错误码
    ├── err_msg                       // 当前错误信息
    ├── confidence                    // 当前定位置信度
    ├── current_station               // 当前所在站点唯一编码
    ├── last_station                  // 上一个所在站点唯一编码
    ├── pose_x                        // 小车 X 坐标
    ├── pose_y                        // 小车 Y 坐标
    ├── angle                         // 小车角度，单位：rad
    ├── vel_x                         // 小车 X 方向速度，单位：m/s
    ├── vel_y                         // 小车 Y 方向速度，单位：m/s
    ├── vel_ang                       // 小车角速度，顺时针为正，单位：rad/s
    ├── battery_temp                  // 电池温度，单位：℃
    ├── battery_level                 // 电池电量，范围：[0, 1]
    ├── charging                      // 是否正在充电，true/false
    ├── voltage                       // 电池电压，单位：V
    ├── current                       // 电池电流，单位：A
    ├── max_charge_voltage            // 允许充电的最大电压，-1 表示不支持，单位：V
    ├── max_charge_current            // 允许充电的最大电流，-1 表示不支持，单位：A
    ├── manual_charge                 // 是否手动充电，true/false
    ├── auto_charge                   // 是否自动充电，true/false
    ├── battery_cycle                 // 电池循环次数
    ├── battery_user_data             // 电池用户自定义数据
    ├── blocked                       // 小车是否被阻挡，true/false
    ├── has_err                       // 小车是否报错，true/false
    ├── map_version                   // 当前使用的地图版本
    │
    └── guide_state                   // 小车导航状态
        ├── task_status               // 导航任务状态 0 = NONE, 1 = WAITING(目前不可能出现该状态), 2 = RUNNING, 3 = SUSPENDED, 4 = COMPLETED, 5 = FAILED, 6 = CANCELED
        ├── task_type                 // 导航任务类型 0 = 没有导航, 1 = 自由导航到任意点, 2 = 自由导航到站点, 3 = 路径导航到站点, 7 = 平动转动, 100 = 其他
        └── target_station            // 导航目标站点唯一编码
```

`command stream`中存储的内容格式，只写入一对filed value，filed为payload，value为json：

```json
// 指令下发
{
    "timestamp": "string",              // 任务下发时间，时间戳形式
    "expiration_time": "string",        // 超时截至时间戳（毫秒）
    "robot_code": "string",             // 调度执行该运单的机器人
    "task_id": "string",                // 任务业务ID（格式：时间戳+短ID，例如 202608180001）
    "sequence": "string",               // 任务中第几个运单
    "waybill_id": "string",             // 运单业务ID（格式：时间戳+短ID，例如 202608180001）
    "waybill_type": "string",           // 下发指令任务类型：move / inspect / load_down / load_up / charge 等，从属运单
    "command_id": "string",             // 下发指令ID（雪花算法生成）
    "command_action": "string",         // 下发动作：dispatch / cancel / pause / resume
    "station_code": "APXX",             // 下发工作站点
    "container_code": "1"               // 操作的货物ID（纯数字）
}
```

```json
    "sequence": "string",               // 任务中第几个运单
    "waybill_id": "string",             // 执行的运单业务ID
    "waybill_type": "string",           // 下发指令任务类型：move / inspect / load_down / load_up / charge 等，从属运单
    "command_id": "string",             // 下发指令ID
    "command_action": "string",         // 下发动作：dispatch / cancel / pause / resume
    "station_code": "APXX",             // 下发工作站点
    "container_code": "1"               // 操作的货物ID（纯数字）

```
